import "C"

import (
	"fmt"
	"path/filepath"
	"unicode/utf16"
	"unsafe"
)

func nativeChooseFiles(multiple bool) ([]string, error) {
	buffer := make([]C.wchar_t, 65536)
	var code C.DWORD
	result := C.dsh_choose_files(C.int(boolInt(multiple)), &buffer[0], C.DWORD(len(buffer)), &code)
	if result == 0 {
		return []string{}, nil
	}
	if result < 0 {
		return nil, fmt.Errorf("open file dialog failed: %d", uint32(code))
	}
	fields := decodeWindowsFileBuffer(buffer)
	if len(fields) < 2 {
		return fields, nil
	}
	paths := make([]string, 0, len(fields)-1)
	for _, name := range fields[1:] {
		paths = append(paths, filepath.Join(fields[0], name))
	}
	return paths, nil
}

func decodeWindowsFileBuffer(buffer []C.wchar_t) []string {
	var fields []string
	for start := 0; start < len(buffer) && buffer[start] != 0; {
		end := start
		for end < len(buffer) && buffer[end] != 0 {
			end++
		}
		value := make([]uint16, end-start)
		for i := range value {
			value[i] = uint16(buffer[start+i])
		}
		fields = append(fields, string(utf16.Decode(value)))
		start = end + 1
	}
	return fields
}

func nativeOpenExternal(raw string) error {
	value, err := validateExternalURL(raw)
	if err != nil {
		return err
	}
	cURL := C.CString(value)
	defer C.free(unsafe.Pointer(cURL))
	if code := int(C.dsh_open_external(cURL)); code != 0 {
		return fmt.Errorf("open external URL failed: %d", code)
	}
	return nil
}
