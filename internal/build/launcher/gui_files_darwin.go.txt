import "C"

import (
	"errors"
	"unsafe"
)

func nativeChooseFiles(multiple bool) ([]string, error) {
	buffer := make([]C.char, 65536)
	result := C.dsh_choose_files(C.int(boolInt(multiple)), &buffer[0], C.size_t(len(buffer)))
	if result == 0 {
		return []string{}, nil
	}
	if result < 0 {
		return nil, errors.New("selected file paths exceed 64 KiB")
	}
	var paths []string
	for start := 0; start < len(buffer) && buffer[start] != 0; {
		end := start
		for end < len(buffer) && buffer[end] != 0 {
			end++
		}
		value := make([]byte, end-start)
		for i := range value {
			value[i] = byte(buffer[start+i])
		}
		paths = append(paths, string(value))
		start = end + 1
	}
	if len(paths) == 0 {
		return nil, errors.New("file dialog returned no paths")
	}
	return paths, nil
}

func nativeOpenExternal(raw string) error {
	value, err := validateExternalURL(raw)
	if err != nil {
		return err
	}
	cURL := C.CString(value)
	defer C.free(unsafe.Pointer(cURL))
	if C.dsh_open_external(cURL) == 0 {
		return errors.New("open external URL failed")
	}
	return nil
}
