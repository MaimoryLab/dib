import "C"

import (
	"errors"
	"os"
	"path/filepath"
	"unsafe"
)

func nativeTrayInit(handle unsafe.Pointer) error {
	executable, err := os.Executable()
	if err != nil {
		return err
	}
	root := filepath.Dir(executable)
	if filepath.Base(root) == "MacOS" {
		root = filepath.Clean(filepath.Join(root, "..", "Resources"))
	}
	iconPath := C.CString(filepath.Join(root, "icon.png"))
	defer C.free(unsafe.Pointer(iconPath))
	if C.dsh_tray_init(handle, iconPath) == 0 {
		return errors.New("create macOS menu-bar icon")
	}
	return nil
}

func nativeTrayClear() { C.dsh_tray_clear() }
