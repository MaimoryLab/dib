package main

import "C"

import (
	"errors"
	"unsafe"
)

func nativeTraySupported() bool { return true }

func nativeTrayAttach(handle unsafe.Pointer) { C.dsh_tray_attach(handle) }

func nativeTraySet(title, tooltip string) error {
	titleValue := C.CString(title)
	tooltipValue := C.CString(tooltip)
	defer C.free(unsafe.Pointer(titleValue))
	defer C.free(unsafe.Pointer(tooltipValue))
	if C.dsh_tray_set(titleValue, tooltipValue) == 0 {
		return errors.New("create macOS menu-bar icon")
	}
	return nil
}

func nativeTrayClear() error {
	if C.dsh_tray_clear() == 0 {
		return errors.New("remove macOS menu-bar icon")
	}
	return nil
}

func nativeTrayShow() error {
	if C.dsh_tray_show() == 0 {
		return errors.New("show dsh window")
	}
	return nil
}

func nativeTrayHide() error {
	if C.dsh_tray_hide() == 0 {
		return errors.New("hide dsh window")
	}
	return nil
}

func nativeTrayTerminate() error {
	if C.dsh_tray_terminate() == 0 {
		return errors.New("close dsh window")
	}
	return nil
}
