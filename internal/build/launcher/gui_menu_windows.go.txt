import "C"

import "unsafe"

func installNativeMenu(handle unsafe.Pointer, version string) {
	value := C.CString(version)
	defer C.free(unsafe.Pointer(value))
	C.dsh_menu_install(handle, value)
}

func uninstallNativeMenu(handle unsafe.Pointer) { C.dsh_menu_uninstall(handle) }
