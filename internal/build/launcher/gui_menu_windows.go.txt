import "C"

import (
	"errors"
	"unsafe"
)

func installNativeMenu(handle unsafe.Pointer, version string) {
	value := C.CString(version)
	defer C.free(unsafe.Pointer(value))
	C.dsh_menu_install(handle, value)
}

func nativeMenuRegister(id, label, parent string, order int) error {
	idValue := C.CString(id)
	labelValue := C.CString(label)
	parentValue := C.CString(parent)
	defer C.free(unsafe.Pointer(idValue))
	defer C.free(unsafe.Pointer(labelValue))
	defer C.free(unsafe.Pointer(parentValue))
	if C.dsh_menu_add(idValue, labelValue, parentValue, C.int(order)) == 0 {
		return errors.New("native menu limit reached")
	}
	return nil
}

func nativeMenuUnregister(id string) error {
	value := C.CString(id)
	defer C.free(unsafe.Pointer(value))
	C.dsh_menu_remove(value)
	return nil
}

func uninstallNativeMenu(handle unsafe.Pointer) { C.dsh_menu_uninstall(handle) }
