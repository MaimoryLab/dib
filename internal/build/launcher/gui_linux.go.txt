import "C"

import (
	"errors"
	"os/exec"
	"runtime"
	"strings"
	"unsafe"
)

type guiWindow struct {
	window *C.DSHWindow
}

func linuxBridgeScript() string {
	var scripts []string
	if _, err := exec.LookPath("xdg-open"); err == nil {
		scripts = append(scripts, "window.dshboxOpenExternal = async url => window.webkit.messageHandlers.dshboxOpenExternal.postMessage(url);")
	}
	if _, err := exec.LookPath("notify-send"); err == nil {
		scripts = append(scripts, "window.dshboxNotify = async request => window.webkit.messageHandlers.dshboxNotify.postMessage(JSON.stringify(request));")
	}
	return strings.Join(scripts, "\n")
}

func newGUIWindow(url string) (*guiWindow, error) {
	bridgeScript := linuxBridgeScript()
	runtime.LockOSThread()
	titleCString := C.CString("DeepSeek Harness")
	urlCString := C.CString(url)
	filesScriptCString := C.CString(droppedFilesScript + "\n" + dshVersionScript())
	bridgeScriptCString := C.CString(bridgeScript)
	defer C.free(unsafe.Pointer(titleCString))
	defer C.free(unsafe.Pointer(urlCString))
	defer C.free(unsafe.Pointer(filesScriptCString))
	defer C.free(unsafe.Pointer(bridgeScriptCString))
	window := C.dsh_window_new(titleCString, urlCString, filesScriptCString, bridgeScriptCString)
	if window == nil {
		runtime.UnlockOSThread()
		return nil, errors.New("initialize GTK4 display")
	}
	return &guiWindow{window: window}, nil
}

func (w *guiWindow) run()       { C.dsh_window_run(w.window) }
func (w *guiWindow) terminate() { C.dsh_window_terminate(w.window) }
func (w *guiWindow) destroy() {
	C.dsh_window_destroy(w.window)
	runtime.UnlockOSThread()
}
