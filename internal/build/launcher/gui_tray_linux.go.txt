package main

import (
	"errors"
	"unsafe"
)

var errNativeTrayUnsupported = errors.New("system tray is unsupported on Linux GTK4 launcher")

func nativeTraySupported() bool          { return false }
func nativeTrayAttach(unsafe.Pointer)    {}
func nativeTraySet(string, string) error { return errNativeTrayUnsupported }
func nativeTrayClear() error             { return errNativeTrayUnsupported }
func nativeTrayShow() error              { return errNativeTrayUnsupported }
func nativeTrayHide() error              { return errNativeTrayUnsupported }
func nativeTrayTerminate() error         { return errNativeTrayUnsupported }
