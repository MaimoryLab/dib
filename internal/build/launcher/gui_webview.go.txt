package main

import (
	"errors"
	"fmt"
	"strings"
	"sync"
	"unicode"

	webview "github.com/webview/webview_go"
)

type guiWindow struct {
	webview.WebView
}

type menuItem struct {
	ID     string `json:"id"`
	Label  string `json:"label"`
	Parent string `json:"parent"`
	Order  int    `json:"order"`
}

type notificationRequest struct {
	Title string `json:"title"`
	Body  string `json:"body"`
}

var activeMenuWindow struct {
	sync.Mutex
	w webview.WebView
}

func newGUIWindow(url string) (*guiWindow, error) {
	w := webview.New(false)
	if w == nil {
		return nil, errors.New("create system WebView")
	}
	w.SetTitle("DeepSeek Harness")
	w.SetSize(1280, 800, webview.HintNone)
	w.Init(inputBehaviorScript + "\n" + droppedFilesScript + "\n" + dshVersionScript())
	w.Navigate(url)
	if err := w.Bind("dshboxNotify", func(request notificationRequest) error {
		if strings.TrimSpace(request.Title) == "" || strings.TrimSpace(request.Body) == "" {
			return errors.New("notification title and body are required")
		}
		if len(request.Title) > 256 || len(request.Body) > 4096 {
			return errors.New("notification exceeds size limit")
		}
		return nativeNotify(request.Title, request.Body)
	}); err != nil {
		w.Destroy()
		return nil, fmt.Errorf("bind notifications: %w", err)
	}
	if err := w.Bind("dshboxChooseFiles", nativeChooseFiles); err != nil {
		w.Destroy()
		return nil, fmt.Errorf("bind file selection: %w", err)
	}
	if err := w.Bind("dshboxOpenExternal", nativeOpenExternal); err != nil {
		w.Destroy()
		return nil, fmt.Errorf("bind external URL opener: %w", err)
	}
	if err := w.Bind("dshboxMenuRegister", func(item menuItem) error {
		item.ID = strings.TrimSpace(item.ID)
		item.Label = strings.TrimSpace(item.Label)
		item.Parent = strings.TrimSpace(item.Parent)
		validText := func(value string, limit int) bool {
			return value != "" && len(value) <= limit && strings.IndexFunc(value, unicode.IsControl) < 0
		}
		if !validText(item.ID, 127) || !validText(item.Label, 256) || len(item.Parent) > 64 || strings.IndexFunc(item.Parent, unicode.IsControl) >= 0 || item.Order < -2147483648 || item.Order > 2147483647 {
			return errors.New("invalid menu item")
		}
		return nativeMenuRegister(item.ID, item.Label, item.Parent, item.Order)
	}); err != nil {
		w.Destroy()
		return nil, fmt.Errorf("bind menu registration: %w", err)
	}
	if err := w.Bind("dshboxMenuUnregister", func(id string) error {
		id = strings.TrimSpace(id)
		if id == "" || len(id) > 127 || strings.IndexFunc(id, unicode.IsControl) >= 0 {
			return errors.New("invalid menu id")
		}
		return nativeMenuUnregister(id)
	}); err != nil {
		w.Destroy()
		return nil, fmt.Errorf("bind menu removal: %w", err)
	}
	if err := nativeTrayInit(w.Window()); err != nil {
		w.Destroy()
		return nil, fmt.Errorf("initialize tray: %w", err)
	}
	installNativeMenu(w.Window(), dshVersion)
	activeMenuWindow.Lock()
	activeMenuWindow.w = w
	activeMenuWindow.Unlock()
	return &guiWindow{WebView: w}, nil
}

func (w *guiWindow) run()       { w.Run() }
func (w *guiWindow) terminate() { w.Terminate() }
func (w *guiWindow) destroy() {
	uninstallNativeMenu(w.Window())
	nativeTrayClear()
	activeMenuWindow.Lock()
	activeMenuWindow.w = nil
	activeMenuWindow.Unlock()
	w.Destroy()
}

func nativeMenuAction(id string) {
	activeMenuWindow.Lock()
	w := activeMenuWindow.w
	activeMenuWindow.Unlock()
	if w == nil {
		return
	}
	w.Eval(fmt.Sprintf("window.dispatchEvent(new CustomEvent('dshbox:menu-select',{detail:%q}))", id))
}
