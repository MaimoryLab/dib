package main

import (
	"fmt"
	"os"
	"os/exec"
	"path/filepath"
	"runtime"
	"strconv"
)

var (
	host       = "127.0.0.1"
	port       = "3080"
	dshPackage = "@deepseek-ai/dsh"
	dshVersion = "unknown"
)

const inputBehaviorScript = `document.documentElement.spellcheck=false;document.documentElement.setAttribute("autocapitalize","off");`

func dshVersionScript() string {
	return "window.dshboxVersion=" + strconv.Quote(dshVersion) + ";"
}

func main() {
	if err := run(); err != nil {
		fmt.Fprintln(os.Stderr, "dshbox:", err)
		os.Exit(1)
	}
}

func serverCommand() (*exec.Cmd, string, error) {
	executable, err := os.Executable()
	if err != nil {
		return nil, "", err
	}
	root := filepath.Dir(executable)
	if runtime.GOOS == "darwin" && filepath.Base(root) == "MacOS" {
		root = filepath.Clean(filepath.Join(root, "..", "Resources"))
	}
	node := filepath.Join(root, "runtime", "node", "bin", "node")
	if runtime.GOOS == "windows" {
		node = filepath.Join(root, "runtime", "node", "node.exe")
	}
	entry := filepath.Join(root, "runtime", "app", "node_modules", filepath.FromSlash(dshPackage), "lib", "bin.js")
	url := "http://" + host + ":" + port
	args := []string{entry, "web"}
	pluginPatch := filepath.Join(root, "runtime", "app", "dsh-desktop.patch.yml")
	if _, err := os.Stat(pluginPatch); err == nil {
		args = append(args, "--patch", pluginPatch)
	}
	args = append(args, "--host", host, "--port", port)
	args = append(args, os.Args[1:]...)
	cmd := exec.Command(node, args...)
	cmd.Stdin = os.Stdin
	cmd.Stdout = os.Stdout
	cmd.Stderr = os.Stderr
	return cmd, url, nil
}
