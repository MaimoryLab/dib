package main

import (
	"errors"
	"fmt"
	"os/exec"
)

func nativeChooseFiles(bool) ([]string, error) {
	return nil, errors.New("choose files: desktop capability unsupported")
}

func nativeOpenExternal(raw string) error {
	value, err := validateExternalURL(raw)
	if err != nil {
		return err
	}
	cmd := exec.Command("xdg-open", value)
	if err := cmd.Start(); err != nil {
		return fmt.Errorf("open external URL: %w", err)
	}
	go func() { _ = cmd.Wait() }()
	return nil
}
