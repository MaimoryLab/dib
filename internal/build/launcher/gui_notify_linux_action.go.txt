package main

/*
#include <stdlib.h>
*/
import "C"

import (
	"encoding/json"
	"strings"
)

type linuxNotificationRequest struct {
	Title string `json:"title"`
	Body  string `json:"body"`
}

//export dshLinuxNotify
func dshLinuxNotify(raw *C.char) {
	var request linuxNotificationRequest
	if err := json.Unmarshal([]byte(C.GoString(raw)), &request); err != nil {
		return
	}
	request.Title = strings.TrimSpace(request.Title)
	request.Body = strings.TrimSpace(request.Body)
	if request.Title == "" || request.Body == "" || len(request.Title) > 256 || len(request.Body) > 4096 {
		return
	}
	go func(title, body string) { _ = nativeNotify(title, body) }(request.Title, request.Body)
}
